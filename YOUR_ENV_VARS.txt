⚠️  IMPORTANT FIREBASE SECURITY STEPS:
───────────────────────────────────────────────────────────────────

After adding variables to Vercel, go to Firebase Console:

1. FIRESTORE RULES:
   https://console.firebase.google.com/project/the-esther-and-mays-group-llc/firestore/rules

   Copy this:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /subcontractors/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    match /{document=**} {
      allow read, write: if request.auth != null &&
        request.auth.token.email == 'estherandmays@gmail.com';
    }
    match /contracts/{contractId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null &&
        request.auth.token.email == 'estherandmays@gmail.com';
    }
  }
}

2. STORAGE RULES:
   https://console.firebase.google.com/project/the-esther-and-mays-group-llc/storage/rules

   Copy this:

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /documents/{userId}/{allPaths=**} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == userId;
    }
  }
}
